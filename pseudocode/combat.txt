# =============================================================================
# CONSTANTES (pourraient venir de settings.py)
# =============================================================================

SEUIL_PV_SWITCH ← 0.3      # 30% de PV restants → déclenche un switch
SEUIL_MULTI_SWITCH ← 0.5    # Multiplicateur < 0.5 → désavantage critique


# =============================================================================
# CLASSE COMBAT
# =============================================================================
#
# Moteur de combat au tour par tour avec IA améliorée.
# Gère la logique d'un combat complet du début à la fin.
# Pas d'affichage ici — que de la logique.
#
# Conforme aux exigences du sujet et avec IA tactique.

CLASSE Combat:

    # -----------------------------------------------------------------
    # CONSTRUCTEUR
    # -----------------------------------------------------------------
    #
    # game_manager : référence vers le jeu, pour accéder au day_night_cycle
    # team_joueur : objet Team du joueur
    # team_adverse : objet Team de l'adversaire (dresseur ou sauvage)
    # type_combat : "wild" ou "trainer"
    # trainer_name : nom du dresseur adverse (None si combat sauvage)
    # trainer_level : niveau du dresseur (B1, B2, B3) pour ajuster l'IA

    MÉTHODE __init__(self, game_manager, team_joueur, team_adverse,
                     type_combat="wild", trainer_name=None, trainer_level=None):

        self.game_manager ← game_manager
        self.team_joueur ← team_joueur
        self.team_adverse ← team_adverse
        self.type_combat ← type_combat
        self.trainer_name ← trainer_name
        self.trainer_level ← trainer_level

        # Pokémon actuellement sur le terrain
        self.pokemon_joueur ← team_joueur.get_premier_valide()
        self.pokemon_adverse ← team_adverse.get_premier_valide()

        # Compteur de tours
        self.tour_actuel ← 0

        # Boosts temporaires (réinitialisés à chaque switch)
        self.boosts_joueur ← {"attack": 0, "defense": 0}
        self.boosts_adverse ← {"attack": 0, "defense": 0}

        # État du combat
        self.combat_termine ← False
        self.resultat_combat ← None
        self.vainqueur ← None
        self.perdant ← None

        # File d'événements du tour
        self.evenements ← liste vide


    # -----------------------------------------------------------------
    # MÉTHODE : executer_tour
    # -----------------------------------------------------------------
    #
    # Exécute un tour complet de combat.
    # Reçoit l'action du joueur, détermine l'action de l'IA,
    # résout les deux actions dans le bon ordre, vérifie les KO.

    MÉTHODE executer_tour(self, action_joueur):
        self.evenements ← liste vide
        self.tour_actuel ← self.tour_actuel + 1

        # --- FUITE ---
        SI action_joueur["type"] == "flee":
            RETOURNER self._gerer_fuite()

        # --- CAPTURE ---
        SI action_joueur["type"] == "capture":
            RETOURNER self._gerer_capture(action_joueur["ball"])

        # --- DÉTERMINER L'ACTION DE L'IA (améliorée) ---
        action_adverse ← self._choisir_action_ia()

        # --- DÉTERMINER L'ORDRE ---
        actions_ordonnees ← self._determiner_ordre(action_joueur, action_adverse)

        # --- EXÉCUTER LES ACTIONS DANS L'ORDRE ---
        POUR CHAQUE (action, camp) DANS actions_ordonnees:
            SI camp == "joueur" ET self.pokemon_joueur.est_ko:
                CONTINUER
            SI camp == "adverse" ET self.pokemon_adverse.est_ko:
                CONTINUER

            self._executer_action(action, camp)

            SI self._verifier_ko():
                SORTIR DE LA BOUCLE

        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _choisir_action_ia
    # -----------------------------------------------------------------
    #
    # IA améliorée : décide tactiquement de l'action adverse.
    # - Combat sauvage : toujours attaquer
    # - Combat dresseur : peut switcher si avantageux
    # - Choisit la meilleure attaque en fonction des types

    MÉTHODE _choisir_action_ia(self):
        # Cas spécial : combat sauvage → toujours attaquer
        SI self.type_combat == "wild":
            meilleure_attaque ← self._trouver_meilleure_attaque(self.pokemon_adverse, self.pokemon_joueur)
            RETOURNER {"type": "attack", "attaque": meilleure_attaque}

        # Combat dresseur : logique tactique
        # 1. Vérifier s'il faut switcher
        SI self._devrait_switcher():
            meilleur_switch ← self._trouver_meilleur_switch()
            SI meilleur_switch EST NOT None:
                index ← self.team_adverse.get_index(meilleur_switch)
                RETOURNER {"type": "switch", "index": index}

        # 2. Sinon, attaquer avec la meilleure attaque
        meilleure_attaque ← self._trouver_meilleure_attaque(self.pokemon_adverse, self.pokemon_joueur)
        RETOURNER {"type": "attack", "attaque": meilleure_attaque}


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _trouver_meilleure_attaque
    # -----------------------------------------------------------------
    #
    # Trouve l'attaque la plus efficace d'un Pokémon contre une cible.
    #
    # attaquant : le Pokémon qui attaque
    # cible : le Pokémon cible
    #
    # Renvoie l'attaque avec le meilleur multiplicateur de type.

    MÉTHODE _trouver_meilleure_attaque(self, attaquant, cible):
        attaques ← attaquant.attaques

        SI len(attaques) == 0:
            RETOURNER {"name": "Lutte", "type": "normal", "power": 30}

        meilleure_attaque ← attaques[0]
        meilleur_multiplicateur ← 0

        POUR CHAQUE attaque DANS attaques:
            multiplicateur ← 1.0

            # Calculer l'efficacité contre chaque type de la cible
            POUR CHAQUE type_cible DANS cible.types:
                mult ← obtenir_multiplicateur(attaque["type"], type_cible)
                multiplicateur ← multiplicateur × mult

            # Si c'est mieux que la meilleure trouvée
            SI multiplicateur > meilleur_multiplicateur:
                meilleur_multiplicateur ← multiplicateur
                meilleure_attaque ← attaque

        RETOURNER meilleure_attaque


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _devrait_switcher
    # -----------------------------------------------------------------
    #
    # Détermine si le dresseur adverse devrait changer de Pokémon.
    #
    # Critères :
    #   1. Pokémon actuel en danger (PV < SEUIL_PV_SWITCH)
    #   2. Pokémon actuel très désavantagé (multiplicateur < SEUIL_MULTI_SWITCH)
    #   3. Il reste des Pokémon valides dans l'équipe
    #
    # Renvoie True si un switch est recommandé.

    MÉTHODE _devrait_switcher(self):
        # Si c'est un combat sauvage, pas de switch
        SI self.type_combat == "wild":
            RETOURNER False

        # Si le dresseur n'a qu'un seul Pokémon valide
        SI self.team_adverse.nombre_valides() <= 1:
            RETOURNER False

        # Critère 1 : PV bas
        SI self.pokemon_adverse.get_pourcentage_pv() < SEUIL_PV_SWITCH:
            RETOURNER True

        # Critère 2 : désavantage de type
        multiplicateur_defense ← self._calculer_multiplicateur_defense()
        SI multiplicateur_defense < SEUIL_MULTI_SWITCH:
            RETOURNER True

        RETOURNER False


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _calculer_multiplicateur_defense
    # -----------------------------------------------------------------
    #
    # Calcule le multiplicateur moyen que subirait le Pokémon adverse
    # si le joueur attaquait avec son type le plus efficace.
    #
    # Utile pour savoir si le Pokémon est en danger.

    MÉTHODE _calculer_multiplicateur_defense(self):
        # On prend le pire scénario : l'attaque la plus efficace du joueur
        meilleure_attaque_joueur ← self._trouver_meilleure_attaque(
            self.pokemon_joueur, self.pokemon_adverse
        )

        multiplicateur ← 1.0
        POUR CHAQUE type_defense DANS self.pokemon_adverse.types:
            mult ← obtenir_multiplicateur(meilleure_attaque_joueur["type"], type_defense)
            multiplicateur ← multiplicateur × mult

        RETOURNER multiplicateur


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _trouver_meilleur_switch
    # -----------------------------------------------------------------
    #
    # Choisit le meilleur Pokémon de remplacement dans l'équipe adverse.
    #
    # Le "meilleur" est celui qui a le meilleur avantage de type
    # contre le Pokémon du joueur.

    MÉTHODE _trouver_meilleur_switch(self):
        meilleur_pokemon ← None
        meilleur_score ← -1

        POUR CHAQUE pokemon DANS self.team_adverse.get_valides():
            # Ne pas se re-switcher sur le même Pokémon
            SI pokemon == self.pokemon_adverse:
                CONTINUER

            # Calculer l'avantage de ce Pokémon contre le joueur
            score ← self._calculer_avantage_type(pokemon, self.pokemon_joueur)

            SI score > meilleur_score:
                meilleur_score ← score
                meilleur_pokemon ← pokemon

        RETOURNER meilleur_pokemon


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _calculer_avantage_type
    # -----------------------------------------------------------------
    #
    # Calcule l'avantage moyen du Pokémon A contre le Pokémon B.
    # Plus le score est élevé, plus A est efficace contre B.

    MÉTHODE _calculer_avantage_type(self, pokemon_a, pokemon_b):
        # On prend la meilleure attaque de A contre B
        meilleure_attaque ← self._trouver_meilleure_attaque(pokemon_a, pokemon_b)

        multiplicateur ← 1.0
        POUR CHAQUE type_b DANS pokemon_b.types:
            mult ← obtenir_multiplicateur(meilleure_attaque["type"], type_b)
            multiplicateur ← multiplicateur × mult

        RETOURNER multiplicateur


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _determiner_ordre
    # -----------------------------------------------------------------
    #
    # Détermine qui agit en premier.
    #
    # Règles de priorité :
    #   1. Les objets et switches passent TOUJOURS en premier
    #   2. Si les deux sont des attaques → ordre aléatoire (50/50)
    #   3. Si les deux sont des objets/switches → le joueur d'abord

    MÉTHODE _determiner_ordre(self, action_joueur, action_adverse):
        priorite_joueur ← action_joueur["type"] DANS ["item", "switch"]
        priorite_adverse ← action_adverse["type"] DANS ["item", "switch"]

        SI priorite_joueur ET priorite_adverse:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]

        SI priorite_joueur:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]

        SI priorite_adverse:
            RETOURNER [(action_adverse, "adverse"), (action_joueur, "joueur")]

        SI random.random() < 0.5:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]
        SINON:
            RETOURNER [(action_adverse, "adverse"), (action_joueur, "joueur")]


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_action
    # -----------------------------------------------------------------
    #
    # Exécute une seule action (attaque, objet ou switch).

    MÉTHODE _executer_action(self, action, camp):
        SI action["type"] == "attack":
            self._executer_attaque(action, camp)

        SINON SI action["type"] == "item":
            self._executer_objet(action, camp)

        SINON SI action["type"] == "switch":
            self._executer_switch(action, camp)


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_attaque
    # -----------------------------------------------------------------
    #
    # Résout une attaque.

    MÉTHODE _executer_attaque(self, action, camp):
        SI camp == "joueur":
            attaquant ← self.pokemon_joueur
            defenseur ← self.pokemon_adverse
            boosts_att ← self.boosts_joueur
            boosts_def ← self.boosts_adverse
        SINON:
            attaquant ← self.pokemon_adverse
            defenseur ← self.pokemon_joueur
            boosts_att ← self.boosts_adverse
            boosts_def ← self.boosts_joueur

        attaque ← action["attaque"]
        day_night ← self.game_manager.day_night_cycle

        resultat ← calculer_degats(
            attaquant, defenseur, attaque, day_night,
            boosts_att, boosts_def
        )

        SI resultat["a_touche"] ET NOT resultat["est_immune"]:
            defenseur.subir_degats(resultat["degats"])

        ajouter {
            "type": "attaque",
            "camp": camp,
            "attaquant": attaquant,
            "defenseur": defenseur,
            "attaque_nom": attaque["name"],
            "resultat": resultat
        } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_objet
    # -----------------------------------------------------------------
    #
    # Résout l'utilisation d'un objet.

    MÉTHODE _executer_objet(self, action, camp):
        item ← action["item"]
        cible ← action.get("cible", self.pokemon_joueur)

        SI item.est_soin:
            cible.soigner(item.effect_value)
            ajouter {"type": "item_utilise", "camp": camp, "item": item,
                     "cible": cible, "effet": "soin", "valeur": item.effect_value} à self.evenements

        SINON SI item.est_boost:
            stat ← item.target_stat
            self.boosts_joueur[stat] ← self.boosts_joueur[stat] + item.effect_value
            ajouter {"type": "item_utilise", "camp": camp, "item": item,
                     "cible": self.pokemon_joueur, "effet": "boost",
                     "stat": stat, "valeur": item.effect_value} à self.evenements

        SINON SI item.est_rappel:
            cible.ranimer(item.effect_value / 100)
            ajouter {"type": "item_utilise", "camp": camp, "item": item,
                     "cible": cible, "effet": "rappel", "valeur": item.effect_value} à self.evenements

        SINON SI item.est_anti_statut:
            cible.soigner_statut()
            ajouter {"type": "item_utilise", "camp": camp, "item": item,
                     "cible": cible, "effet": "anti_statut"} à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_switch
    # -----------------------------------------------------------------
    #
    # Change le Pokémon actif.

    MÉTHODE _executer_switch(self, action, camp):
        SI camp == "joueur":
            ancien ← self.pokemon_joueur
            index ← action["index"]
            nouveau ← self.team_joueur.get_pokemon(index)

            SI nouveau EST NOT None ET NOT nouveau.est_ko:
                self.pokemon_joueur ← nouveau
                self.boosts_joueur ← {"attack": 0, "defense": 0}
                ajouter {"type": "switch", "camp": camp, "ancien": ancien,
                         "nouveau": nouveau} à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _gerer_fuite
    # -----------------------------------------------------------------
    #
    # Gère la fuite du joueur.

    MÉTHODE _gerer_fuite(self):
        SI self.type_combat != "wild":
            ajouter {"type": "fuite_impossible",
                     "message": "Impossible de fuir un combat de dresseur !"} à self.evenements
            RETOURNER self.evenements

        self.combat_termine ← True
        self.resultat_combat ← "fuite"
        ajouter {"type": "fuite", "message": "Vous avez pris la fuite !"} à self.evenements
        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _gerer_capture
    # -----------------------------------------------------------------
    #
    # Gère une tentative de capture.

    MÉTHODE _gerer_capture(self, ball):
        resultat ← tenter_capture(
            self.pokemon_adverse, ball,
            self.team_joueur, self.game_manager.player.storage,
            self.type_combat
        )

        ajouter {"type": "capture", "resultat": resultat, "ball": ball} à self.evenements

        SI resultat["capture_reussie"]:
            self.combat_termine ← True
            self.resultat_combat ← "capture"
        SINON:
            SI resultat["raison_echec"] == "miss":
                action_adverse ← self._choisir_action_ia()
                self._executer_action(action_adverse, "adverse")
                self._verifier_ko()

        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _verifier_ko
    # -----------------------------------------------------------------
    #
    # Vérifie si un Pokémon est KO et gère les conséquences.

    MÉTHODE _verifier_ko(self):
        # --- Pokémon adverse KO ---
        SI self.pokemon_adverse.est_ko:
            ajouter {"type": "ko", "camp": "adverse", "pokemon": self.pokemon_adverse} à self.evenements
            self._donner_xp()

            SI self.team_adverse.a_pokemon_valide:
                suivant ← self.team_adverse.get_premier_valide()
                ancien ← self.pokemon_adverse
                self.pokemon_adverse ← suivant
                self.boosts_adverse ← {"attack": 0, "defense": 0}
                ajouter {"type": "switch", "camp": "adverse", "ancien": ancien,
                         "nouveau": suivant} à self.evenements
            SINON:
                self._terminer_combat("victoire")
                RETOURNER True

        # --- Pokémon du joueur KO ---
        SI self.pokemon_joueur.est_ko:
            ajouter {"type": "ko", "camp": "joueur", "pokemon": self.pokemon_joueur} à self.evenements

            SI self.team_joueur.a_pokemon_valide:
                ajouter {"type": "choix_remplacement", "camp": "joueur"} à self.evenements
                RETOURNER True
            SINON:
                self._terminer_combat("defaite")
                RETOURNER True

        RETOURNER False


    # -----------------------------------------------------------------
    # MÉTHODE : remplacer_pokemon_joueur
    # -----------------------------------------------------------------
    #
    # Appelée par state_combat.py quand le joueur a choisi un remplaçant.

    MÉTHODE remplacer_pokemon_joueur(self, index):
        ancien ← self.pokemon_joueur
        nouveau ← self.team_joueur.get_pokemon(index)

        SI nouveau EST None OU nouveau.est_ko:
            RETOURNER None

        self.pokemon_joueur ← nouveau
        self.boosts_joueur ← {"attack": 0, "defense": 0}

        evenement ← {"type": "switch", "camp": "joueur", "ancien": ancien, "nouveau": nouveau}
        RETOURNER evenement


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _donner_xp
    # -----------------------------------------------------------------
    #
    # Donne l'XP au Pokémon du joueur qui a porté le coup final.

    MÉTHODE _donner_xp(self):
        SI self.pokemon_joueur.est_ko:
            RETOURNER

        xp ← calculer_xp_gagnee(self.pokemon_adverse)
        resultat_xp ← self.pokemon_joueur.gagner_xp(xp)

        ajouter {"type": "xp", "pokemon": self.pokemon_joueur,
                 "xp_gagnee": xp, "resultat": resultat_xp} à self.evenements

        SI resultat_xp["level_ups"] > 0:
            ajouter {"type": "level_up", "pokemon": self.pokemon_joueur,
                     "ancien_niveau": resultat_xp["ancien_niveau"],
                     "nouveau_niveau": resultat_xp["nouveau_niveau"]} à self.evenements

        SI resultat_xp["evolution"]:
            ajouter {"type": "evolution_possible", "pokemon": self.pokemon_joueur} à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _terminer_combat
    # -----------------------------------------------------------------
    #
    # Marque le combat comme terminé.

    MÉTHODE _terminer_combat(self, resultat):
        self.combat_termine ← True
        self.resultat_combat ← resultat

        SI resultat == "victoire":
            self.vainqueur ← self.pokemon_joueur
            self.perdant ← self.pokemon_adverse
        SINON:
            self.vainqueur ← self.pokemon_adverse
            self.perdant ← self.pokemon_joueur

        ajouter {"type": "fin_combat", "resultat": resultat} à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE : get_etat
    # -----------------------------------------------------------------
    #
    # Renvoie l'état actuel du combat pour l'affichage.

    MÉTHODE get_etat(self):
        RETOURNER {
            "pokemon_joueur": self.pokemon_joueur,
            "pokemon_adverse": self.pokemon_adverse,
            "boosts_joueur": copie de self.boosts_joueur,
            "boosts_adverse": copie de self.boosts_adverse,
            "tour": self.tour_actuel,
            "type_combat": self.type_combat,
            "trainer_name": self.trainer_name,
            "combat_termine": self.combat_termine,
            "resultat": self.resultat_combat
        }


    # -----------------------------------------------------------------
    # MÉTHODE : get_vainqueur
    # -----------------------------------------------------------------
    #
    # Renvoie le nom du vainqueur (méthode imposée par le sujet).

    MÉTHODE get_vainqueur(self):
        SI NOT self.combat_termine:
            RETOURNER None
        SI self.vainqueur EST NOT None:
            RETOURNER self.vainqueur.nom
        RETOURNER None


    # -----------------------------------------------------------------
    # MÉTHODE : get_perdant
    # -----------------------------------------------------------------
    #
    # Renvoie le nom du perdant (méthode imposée par le sujet).

    MÉTHODE get_perdant(self):
        SI NOT self.combat_termine:
            RETOURNER None
        SI self.perdant EST NOT None:
            RETOURNER self.perdant.nom
        RETOURNER None


    # -----------------------------------------------------------------
    # MÉTHODE : enregistrer_pokemon_rencontre
    # -----------------------------------------------------------------
    #
    # Enregistre le Pokémon adverse dans le Pokédex (méthode imposée).

    MÉTHODE enregistrer_pokemon_rencontre(self, pokedex):
        SI self.pokemon_adverse EST NOT None:
            pokedex.enregistrer_vu(self.pokemon_adverse.id)
            RETOURNER True
        RETOURNER False


    # -----------------------------------------------------------------
    # MÉTHODE SPÉCIALE : __str__
    # -----------------------------------------------------------------

    MÉTHODE __str__(self):
        RETOURNER "Combat tour {} : {} (PV:{}/{}) vs {} (PV:{}/{})".format(
            self.tour_actuel,
            self.pokemon_joueur.nom, self.pokemon_joueur.pv_actuels, self.pokemon_joueur.pv_max,
            self.pokemon_adverse.nom, self.pokemon_adverse.pv_actuels, self.pokemon_adverse.pv_max
        )

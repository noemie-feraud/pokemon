importer calculer_degats, calculer_xp_gagnee depuis combat.damage_calculator

# =============================================================================
# CLASSE COMBAT
# =============================================================================
#
# Moteur de combat au tour par tour.
# Gère la logique d'un combat complet du début à la fin.
# Pas d'affichage ici — que de la logique.

CLASSE Combat:

    # -----------------------------------------------------------------
    # CONSTRUCTEUR
    # -----------------------------------------------------------------
    #
    # game_manager : référence vers le jeu, pour accéder au day_night_cycle
    # team_joueur : objet Team du joueur
    # team_adverse : objet Team de l'adversaire (dresseur ou sauvage)
    # type_combat : "wild" ou "trainer"
    # trainer_name : nom du dresseur adverse (None si combat sauvage)
    #
    # Le combat commence avec le premier Pokémon valide de chaque équipe
    # sur le terrain.
    #
    # tour_actuel : compteur de tours pour le debug et les stats.
    #
    # boosts : les bonus temporaires de stats pour chaque camp.
    #   Réinitialisés quand le Pokémon actif change.
    #
    # combat_termine : flag pour savoir si le combat est fini.
    #   Quand c'est True, state_combat.py sait qu'il doit se pop.
    #
    # resultat_combat : "victoire", "defaite", "capture", "fuite", ou None.
    #
    # derniere_action : stocke le résultat de la dernière action pour
    #   que state_combat.py puisse l'afficher.

    MÉTHODE __init__(self, game_manager, team_joueur, team_adverse,
                     type_combat="wild", trainer_name=None):

        self.game_manager ← game_manager
        self.team_joueur ← team_joueur
        self.team_adverse ← team_adverse
        self.type_combat ← type_combat
        self.trainer_name ← trainer_name

        # Pokémon actuellement sur le terrain
        self.pokemon_joueur ← team_joueur.get_premier_valide()
        self.pokemon_adverse ← team_adverse.get_premier_valide()

        # Compteur de tours
        self.tour_actuel ← 0

        # Boosts temporaires (réinitialisés à chaque switch)
        self.boosts_joueur ← {"attack": 0, "defense": 0}
        self.boosts_adverse ← {"attack": 0, "defense": 0}

        # État du combat
        self.combat_termine ← False
        self.resultat_combat ← None

        # File d'événements du tour
        # Chaque action produit une liste d'événements que state_combat.py
        # consomme un par un pour l'affichage séquentiel.
        self.evenements ← liste vide


    # -----------------------------------------------------------------
    # MÉTHODE : executer_tour
    # -----------------------------------------------------------------
    #
    # Exécute un tour complet de combat.
    # Reçoit l'action du joueur, détermine l'action de l'IA,
    # résout les deux actions dans le bon ordre, vérifie les KO.
    #
    # action_joueur : dictionnaire décrivant ce que le joueur fait :
    #   {"type": "attack", "attaque": {...}}
    #   {"type": "item", "item": objet Item, "cible": objet Pokemon}
    #   {"type": "switch", "index": int}
    #   {"type": "flee"}
    #   {"type": "capture", "ball": objet Item}
    #
    # Renvoie une liste d'événements dans l'ordre chronologique.
    # Chaque événement est un dictionnaire :
    #   {"type": "attaque", "attaquant": ..., "defenseur": ..., "resultat": ...}
    #   {"type": "ko", "pokemon": ...}
    #   {"type": "switch", "ancien": ..., "nouveau": ...}
    #   {"type": "item_utilise", "item": ..., "cible": ...}
    #   {"type": "capture", "resultat": ...}
    #   {"type": "fuite"}
    #   {"type": "fin_combat", "resultat": ...}
    #   {"type": "xp", "pokemon": ..., "xp_gagnee": ..., "resultat_xp": ...}
    #
    # state_combat.py parcourt ces événements et les affiche un par un
    # avec les animations et messages correspondants.

    MÉTHODE executer_tour(self, action_joueur):
        self.evenements ← liste vide
        self.tour_actuel ← self.tour_actuel + 1

        # --- FUITE ---
        SI action_joueur["type"] == "flee":
            RETOURNER self._gerer_fuite()

        # --- CAPTURE ---
        SI action_joueur["type"] == "capture":
            RETOURNER self._gerer_capture(action_joueur["ball"])

        # --- DÉTERMINER L'ACTION DE L'IA ---
        action_adverse ← self._choisir_action_ia()

        # --- DÉTERMINER L'ORDRE ---
        # Les objets et switches passent toujours en premier.
        # Sinon, ordre aléatoire.
        actions_ordonnees ← self._determiner_ordre(action_joueur, action_adverse)

        # --- EXÉCUTER LES ACTIONS DANS L'ORDRE ---
        POUR CHAQUE (action, camp) DANS actions_ordonnees:
            # Vérifier que le combattant est encore en vie
            # (il a pu être KO par l'action précédente)
            SI camp == "joueur" ET self.pokemon_joueur.est_ko:
                CONTINUER
            SI camp == "adverse" ET self.pokemon_adverse.est_ko:
                CONTINUER

            self._executer_action(action, camp)

            # Vérifier les KO après chaque action
            SI self._verifier_ko():
                # Le combat est terminé ou un remplacement est nécessaire
                SORTIR DE LA BOUCLE

        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _choisir_action_ia
    # -----------------------------------------------------------------
    #
    # L'IA choisit une action pour le Pokémon adverse.
    # Pour l'instant : attaque aléatoire parmi les attaques disponibles.
    #
    # Si on veut une IA plus intelligente plus tard, c'est ici qu'on
    # la met. On pourrait analyser les types et choisir l'attaque la
    # plus efficace. Mais pour le MVP, l'aléatoire suffit.

    MÉTHODE _choisir_action_ia(self):
        attaques ← self.pokemon_adverse.attaques

        SI attaques EST vide:
            # Cas de sécurité : pas d'attaque → "Lutte" (attaque de base)
            RETOURNER {
                "type": "attack",
                "attaque": {"name": "Lutte", "type": "normal", "power": 30}
            }

        attaque_choisie ← random.choice(attaques)
        RETOURNER {"type": "attack", "attaque": attaque_choisie}


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _determiner_ordre
    # -----------------------------------------------------------------
    #
    # Détermine qui agit en premier.
    #
    # Règles de priorité :
    #   1. Les objets et switches passent TOUJOURS en premier
    #   2. Si les deux sont des attaques → ordre aléatoire (50/50)
    #   3. Si les deux sont des objets/switches → le joueur d'abord
    #
    # Renvoie une liste de tuples (action, camp) dans l'ordre.

    MÉTHODE _determiner_ordre(self, action_joueur, action_adverse):
        priorite_joueur ← action_joueur["type"] DANS ["item", "switch"]
        priorite_adverse ← action_adverse["type"] DANS ["item", "switch"]

        # Les deux sont prioritaires → joueur d'abord
        SI priorite_joueur ET priorite_adverse:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]

        # Seul le joueur est prioritaire
        SI priorite_joueur:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]

        # Seul l'adversaire est prioritaire
        SI priorite_adverse:
            RETOURNER [(action_adverse, "adverse"), (action_joueur, "joueur")]

        # Les deux attaquent → aléatoire
        SI random.random() < 0.5:
            RETOURNER [(action_joueur, "joueur"), (action_adverse, "adverse")]
        SINON:
            RETOURNER [(action_adverse, "adverse"), (action_joueur, "joueur")]


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_action
    # -----------------------------------------------------------------
    #
    # Exécute une seule action (attaque, objet ou switch).
    # Ajoute les événements correspondants à self.evenements.
    #
    # action : le dictionnaire de l'action
    # camp : "joueur" ou "adverse"

    MÉTHODE _executer_action(self, action, camp):
        SI action["type"] == "attack":
            self._executer_attaque(action, camp)

        SINON SI action["type"] == "item":
            self._executer_objet(action, camp)

        SINON SI action["type"] == "switch":
            self._executer_switch(action, camp)


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_attaque
    # -----------------------------------------------------------------
    #
    # Résout une attaque.
    #
    # 1. Identifier l'attaquant et le défenseur
    # 2. Appeler damage_calculator pour obtenir les dégâts
    # 3. Appliquer les dégâts au défenseur
    # 4. Ajouter les événements

    MÉTHODE _executer_attaque(self, action, camp):
        SI camp == "joueur":
            attaquant ← self.pokemon_joueur
            defenseur ← self.pokemon_adverse
            boosts_att ← self.boosts_joueur
            boosts_def ← self.boosts_adverse
        SINON:
            attaquant ← self.pokemon_adverse
            defenseur ← self.pokemon_joueur
            boosts_att ← self.boosts_adverse
            boosts_def ← self.boosts_joueur

        attaque ← action["attaque"]

        # Récupérer le cycle jour/nuit (peut être None)
        day_night ← self.game_manager.day_night_cycle

        # Calculer les dégâts
        resultat ← calculer_degats(
            attaquant, defenseur, attaque, day_night,
            boosts_att, boosts_def
        )

        # Appliquer les dégâts
        SI resultat["a_touche"] ET NOT resultat["est_immune"]:
            defenseur.subir_degats(resultat["degats"])

        # Ajouter l'événement
        ajouter {
            "type": "attaque",
            "camp": camp,
            "attaquant": attaquant,
            "defenseur": defenseur,
            "attaque_nom": attaque["name"],
            "resultat": resultat
        } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_objet
    # -----------------------------------------------------------------
    #
    # Résout l'utilisation d'un objet par le joueur.
    # L'IA n'utilise pas d'objets (simplification).
    #
    # Les types d'objets :
    #   - Soin → restaure des PV à la cible
    #   - Boost → augmente une stat temporairement
    #   - Rappel → réanime un Pokémon KO
    #   - Anti-statut → retire un statut

    MÉTHODE _executer_objet(self, action, camp):
        item ← action["item"]
        cible ← action.get("cible", self.pokemon_joueur)

        SI item.est_soin:
            cible.soigner(item.effect_value)
            ajouter {
                "type": "item_utilise",
                "camp": camp,
                "item": item,
                "cible": cible,
                "effet": "soin",
                "valeur": item.effect_value
            } à self.evenements

        SINON SI item.est_boost:
            # Appliquer le boost au camp du joueur
            stat ← item.target_stat    # "attack" ou "defense"
            self.boosts_joueur[stat] ← self.boosts_joueur[stat] + item.effect_value

            ajouter {
                "type": "item_utilise",
                "camp": camp,
                "item": item,
                "cible": self.pokemon_joueur,
                "effet": "boost",
                "stat": stat,
                "valeur": item.effect_value
            } à self.evenements

        SINON SI item.est_rappel:
            cible.ranimer(item.effect_value / 100)
            ajouter {
                "type": "item_utilise",
                "camp": camp,
                "item": item,
                "cible": cible,
                "effet": "rappel",
                "valeur": item.effect_value
            } à self.evenements

        SINON SI item.est_anti_statut:
            cible.soigner_statut()
            ajouter {
                "type": "item_utilise",
                "camp": camp,
                "item": item,
                "cible": cible,
                "effet": "anti_statut"
            } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _executer_switch
    # -----------------------------------------------------------------
    #
    # Change le Pokémon actif du joueur.
    # Réinitialise les boosts (comme dans les vrais Pokémon).
    #
    # action["index"] : l'index du Pokémon à envoyer dans l'équipe.

    MÉTHODE _executer_switch(self, action, camp):
        SI camp == "joueur":
            ancien ← self.pokemon_joueur
            index ← action["index"]
            nouveau ← self.team_joueur.get_pokemon(index)

            SI nouveau EST NOT None ET NOT nouveau.est_ko:
                self.pokemon_joueur ← nouveau

                # Réinitialiser les boosts
                self.boosts_joueur ← {"attack": 0, "defense": 0}

                ajouter {
                    "type": "switch",
                    "camp": camp,
                    "ancien": ancien,
                    "nouveau": nouveau
                } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _gerer_fuite
    # -----------------------------------------------------------------
    #
    # Gère la fuite du joueur.
    # Possible uniquement en combat sauvage.
    # Toujours réussie (pas de probabilité d'échec).
    #
    # Renvoie les événements.

    MÉTHODE _gerer_fuite(self):
        SI self.type_combat != "wild":
            ajouter {
                "type": "fuite_impossible",
                "message": "Impossible de fuir un combat de dresseur !"
            } à self.evenements
            RETOURNER self.evenements

        self.combat_termine ← True
        self.resultat_combat ← "fuite"

        ajouter {
            "type": "fuite",
            "message": "Vous avez pris la fuite !"
        } à self.evenements

        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _gerer_capture
    # -----------------------------------------------------------------
    #
    # Gère une tentative de capture.
    # Appelle capture_system.tenter_capture().
    #
    # ball : l'objet Item (la Pokéball choisie)

    MÉTHODE _gerer_capture(self, ball):
        resultat ← tenter_capture(
            self.pokemon_adverse, ball,
            self.team_joueur, self.game_manager.player.storage,
            self.type_combat
        )

        ajouter {
            "type": "capture",
            "resultat": resultat,
            "ball": ball
        } à self.evenements

        SI resultat["capture_reussie"]:
            self.combat_termine ← True
            self.resultat_combat ← "capture"
        # Si capture ratée, le tour continue avec l'attaque de l'adversaire
        SINON:
            SI resultat["raison_echec"] == "miss":
                # L'adversaire attaque après la capture ratée
                action_adverse ← self._choisir_action_ia()
                self._executer_action(action_adverse, "adverse")
                self._verifier_ko()

        RETOURNER self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _verifier_ko
    # -----------------------------------------------------------------
    #
    # Vérifie si un Pokémon est KO après une action.
    # Gère les conséquences : remplacement ou fin de combat.
    #
    # Renvoie True si le combat est terminé ou si un remplacement
    # est nécessaire, False si le combat continue normalement.

    MÉTHODE _verifier_ko(self):
        # --- Pokémon adverse KO ---
        SI self.pokemon_adverse.est_ko:
            ajouter {
                "type": "ko",
                "camp": "adverse",
                "pokemon": self.pokemon_adverse
            } à self.evenements

            # Donner l'XP au Pokémon du joueur
            self._donner_xp()

            # Vérifier s'il reste des Pokémon adverses
            SI self.team_adverse.a_pokemon_valide:
                # Envoyer le suivant
                suivant ← self.team_adverse.get_premier_valide()
                ancien ← self.pokemon_adverse
                self.pokemon_adverse ← suivant
                self.boosts_adverse ← {"attack": 0, "defense": 0}

                ajouter {
                    "type": "switch",
                    "camp": "adverse",
                    "ancien": ancien,
                    "nouveau": suivant
                } à self.evenements
            SINON:
                # Plus de Pokémon adverses → victoire !
                self._terminer_combat("victoire")
                RETOURNER True

        # --- Pokémon du joueur KO ---
        SI self.pokemon_joueur.est_ko:
            ajouter {
                "type": "ko",
                "camp": "joueur",
                "pokemon": self.pokemon_joueur
            } à self.evenements

            SI self.team_joueur.a_pokemon_valide:
                # Le joueur doit choisir un remplaçant
                # On signale à state_combat.py qu'il faut un choix
                ajouter {
                    "type": "choix_remplacement",
                    "camp": "joueur"
                } à self.evenements
                RETOURNER True
            SINON:
                # Plus de Pokémon → défaite
                self._terminer_combat("defaite")
                RETOURNER True

        RETOURNER False


    # -----------------------------------------------------------------
    # MÉTHODE : remplacer_pokemon_joueur
    # -----------------------------------------------------------------
    #
    # Appelée par state_combat.py quand le joueur a choisi un
    # remplaçant après un KO.
    #
    # index : l'index du Pokémon choisi dans l'équipe.
    #
    # Renvoie l'événement de switch pour l'affichage.

    MÉTHODE remplacer_pokemon_joueur(self, index):
        ancien ← self.pokemon_joueur
        nouveau ← self.team_joueur.get_pokemon(index)

        SI nouveau EST None OU nouveau.est_ko:
            RETOURNER None

        self.pokemon_joueur ← nouveau
        self.boosts_joueur ← {"attack": 0, "defense": 0}

        evenement ← {
            "type": "switch",
            "camp": "joueur",
            "ancien": ancien,
            "nouveau": nouveau
        }

        RETOURNER evenement


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _donner_xp
    # -----------------------------------------------------------------
    #
    # Donne l'XP au Pokémon du joueur qui a porté le coup final.
    # Appelée quand un Pokémon adverse est mis KO.
    #
    # L'XP est donnée uniquement au Pokémon actif du joueur
    # (celui qui a combattu). Pas de partage d'XP.
    #
    # On appelle pokemon.gagner_xp() qui gère le level up en interne
    # et renvoie les infos (nombre de niveaux gagnés, possibilité
    # d'évolution).
    #
    # On ajoute les événements d'XP et de level up pour l'affichage.

    MÉTHODE _donner_xp(self):
        SI self.pokemon_joueur.est_ko:
            RETOURNER    # un Pokémon KO gagne pas d'XP

        xp ← calculer_xp_gagnee(self.pokemon_adverse)
        resultat_xp ← self.pokemon_joueur.gagner_xp(xp)

        ajouter {
            "type": "xp",
            "pokemon": self.pokemon_joueur,
            "xp_gagnee": xp,
            "resultat": resultat_xp
        } à self.evenements

        # Si level up → événement supplémentaire
        SI resultat_xp["level_ups"] > 0:
            ajouter {
                "type": "level_up",
                "pokemon": self.pokemon_joueur,
                "ancien_niveau": resultat_xp["ancien_niveau"],
                "nouveau_niveau": resultat_xp["nouveau_niveau"]
            } à self.evenements

        # Si évolution possible → événement supplémentaire
        SI resultat_xp["evolution"]:
            ajouter {
                "type": "evolution_possible",
                "pokemon": self.pokemon_joueur
            } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE PRIVÉE : _terminer_combat
    # -----------------------------------------------------------------
    #
    # Marque le combat comme terminé et ajoute l'événement de fin.
    #
    # resultat : "victoire" ou "defaite"

    MÉTHODE _terminer_combat(self, resultat):
        self.combat_termine ← True
        self.resultat_combat ← resultat

        ajouter {
            "type": "fin_combat",
            "resultat": resultat
        } à self.evenements


    # -----------------------------------------------------------------
    # MÉTHODE : get_etat
    # -----------------------------------------------------------------
    #
    # Renvoie un snapshot de l'état actuel du combat.
    # Utilisé par state_combat.py et combat_ui.py pour l'affichage.
    #
    # Toutes les infos nécessaires pour dessiner l'écran de combat
    # sont là : les deux Pokémon, leurs PV, les boosts, le tour, etc.

    MÉTHODE get_etat(self):
        RETOURNER {
            "pokemon_joueur": self.pokemon_joueur,
            "pokemon_adverse": self.pokemon_adverse,
            "boosts_joueur": copie de self.boosts_joueur,
            "boosts_adverse": copie de self.boosts_adverse,
            "tour": self.tour_actuel,
            "type_combat": self.type_combat,
            "trainer_name": self.trainer_name,
            "combat_termine": self.combat_termine,
            "resultat": self.resultat_combat
        }


    # -----------------------------------------------------------------
    # MÉTHODE SPÉCIALE : __str__
    # -----------------------------------------------------------------

    MÉTHODE __str__(self):
        RETOURNER "Combat tour {} : {} (PV:{}/{}) vs {} (PV:{}/{})".format(
            self.tour_actuel,
            self.pokemon_joueur.nom, self.pokemon_joueur.pv_actuels, self.pokemon_joueur.pv_max,
            self.pokemon_adverse.nom, self.pokemon_adverse.pv_actuels, self.pokemon_adverse.pv_max
        )
